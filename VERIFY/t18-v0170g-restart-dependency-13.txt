import assert from 'node:assert/strict';
import { readFile, writeFile, readdir, access } from 'node:fs/promises';
import { createHash } from 'node:crypto';
import { join } from 'node:path';
import { verifyFrozenT18Labels } from './t18-label-preservation.mjs';
import { verifyRestartStateScope } from './t18-restart-state-scope.mjs';

const directory = 'C:/Users/a1500/AppData/Local/Temp/router-implementation';
const notes = join(directory, 'implement-spec-go-20261009');
const hash = bytes => createHash('sha256').update(bytes).digest('hex').toUpperCase();
const label = process.argv[2], phase = process.argv[3];
assert(/^t18-v017\d[a-z]$/.test(label));
assert(['failed-controlled', 'failed-restart', 'failed-postprocessing', 'validation-complete-pending-independent-final-review'].includes(phase));
const failed = phase.startsWith('failed');
const target = join(directory, label + '-' + (failed ? 'failed' : 'validation') + '-frozen.json');
try { await access(target); throw new Error('Frozen capture already exists'); }
catch (error) { if (error.code !== 'ENOENT') throw error; }
const priorLabels = await verifyFrozenT18Labels();
const paths = new Set((await readdir(directory, { withFileTypes: true })).filter(entry => entry.isFile()
  && entry.name.startsWith(label + '-') && !entry.name.endsWith('-frozen.json')).map(entry => join(directory, entry.name)));
for (const record of priorLabels.records) paths.add(join(directory, record.manifest));
for (const suffix of ['controlled-execution-sources.json', 'restart-execution-sources.json']) {
  try {
    const sources = JSON.parse(await readFile(join(directory, label + '-' + suffix), 'utf8'));
    for (const entry of sources.copied) {
      const bytes = await readFile(entry.executed); assert.equal(bytes.length, entry.bytes); assert.equal(hash(bytes), entry.sha256);
      paths.add(entry.executed);
    }
    for (const entry of sources.dependencies) {
      const bytes = await readFile(entry.captured); assert.equal(bytes.length, entry.bytes); assert.equal(hash(bytes), entry.sha256);
      if (!failed) assert.equal(hash(await readFile(entry.path)), entry.sha256, 'An executed dependency changed after capture');
      paths.add(entry.captured);
    }
  } catch (error) { if (error.code !== 'ENOENT') throw error; }
}
for (const path of [join(directory, label + '-controlled-home/t18-controlled-counts.json'),
  join(directory, label + '-controlled-home/t18-controlled-scenario.json'),
  join(directory, label + '-fixture-bundle/lib/index.js'), join(directory, label + '-controlled-home/router/desktop/state.json')]) {
  try { await access(path); paths.add(path); } catch (error) { if (error.code !== 'ENOENT') throw error; }
}
if (!failed) {
  const manifest = JSON.parse(await readFile(join(directory, label + '-package-identity.json'), 'utf8'));
  assert.equal(hash(await readFile(manifest.path)), manifest.sha256); paths.add(manifest.path);
  for (const entry of manifest.files) {
    const path = join(directory, label + '-controlled-home/profiles/desktop/node_modules/@irishwei/dsh-router', entry.file);
    const bytes = await readFile(path); assert.equal(bytes.length, entry.bytes); assert.equal(hash(bytes), entry.sha256);
    paths.add(path);
  }
  assert.deepEqual(await verifyRestartStateScope(label), JSON.parse(await readFile(join(directory, label + '-restart-state-scope.json'), 'utf8')));
  for (const entry of await readdir(notes, { withFileTypes: true })) {
    if (entry.isFile() && (/^t18-(?:slice|source|r[0-9]+-fix|merged|merger|review|pack-reviewed).*\.(?:log|md|json|diff|patch|tar|ps1|mjs)$/.test(entry.name)
      || entry.name === 't18-final-reviewed-source.json')) paths.add(join(notes, entry.name));
  }
}
const files = [];
for (const path of [...paths].sort()) { const bytes = await readFile(path); files.push({ path, bytes: bytes.length, sha256: hash(bytes) }); }
assert(files.length > 0);
await writeFile(target, JSON.stringify({ label, phase, files, realModelRequests: 0, priorLabels,
  note: 'Raw owned receipts and executed source copies retain their original phase. Later independent review uses a separate archive index.' }, null, 2), { flag: 'wx' });
console.log(JSON.stringify({ labelFrozen: true, label, phase, files: files.length, realModelRequests: 0 }));
