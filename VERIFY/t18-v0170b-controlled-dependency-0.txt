import assert from 'node:assert/strict';
import { readFile, writeFile } from 'node:fs/promises';
import { createHash } from 'node:crypto';
import { join } from 'node:path';
import { verifyFrozenT18Labels } from './t18-label-preservation.mjs';

const directory = 'C:/Users/a1500/AppData/Local/Temp/router-implementation';
const notes = join(directory, 'implement-spec-go-20261009');
const stage = process.argv[2], label = process.argv[3];
assert(['controlled', 'restart'].includes(stage));
assert(/^t18-v017\d[a-z]$/.test(label));
await verifyFrozenT18Labels();
const hash = bytes => createHash('sha256').update(bytes).digest('hex').toUpperCase();
const copied = [];
for (const [source, suffix] of [['t18-desktop-rpc-draft.mjs', 'rpc.mjs'], ['t18-installed-renderer-draft.mjs', 'renderer.mjs']]) {
  const path = join(notes, source), bytes = await readFile(path);
  const executed = join(notes, label + '-' + stage + '-executed-' + suffix);
  await writeFile(executed, bytes, { flag: 'wx' });
  copied.push({ source: path, executed, bytes: bytes.length, sha256: hash(bytes) });
}
const dependencies = [];
const paths = [
  't18-capture-execution-sources.mjs', 't18-create-validation-plan.mjs', 't18-run-controlled.ps1', 't18-preservation.mjs', 't18-label-preservation.mjs',
  't18-desktop-prepare.mjs', 't18-package-freeze.mjs', 't18-pack-reviewed.ps1', 't18-fixture-final.mjs',
  't18-state-seal.mjs', 't18-restart-state-scope.mjs', 't18-export-proof.mjs', 't18-public-summary.mjs', 't18-freeze-label.mjs',
  't18-source-review-issue-1.json', 't18-source-review-issue-19.json', 't18-development-authority.json', 't18-final-reviewed-source.json',
  label + '-validation-plan.json',
].map(name => join(notes, name));
paths.push(join('E:/GPT/Router项目', 'test/client-harness.mjs'),
  join(directory, label + '-reviewed-contract.md'), join(directory, label + '-host-rpc.mjs'),
  join(directory, label + '-desktop-owned.ps1'), join(directory, label + '-install.ps1'));
for (const path of paths) {
  const bytes = await readFile(path);
  const captured = join(notes, label + '-' + stage + '-dependency-' + dependencies.length + '.txt');
  await writeFile(captured, bytes, { flag: 'wx' });
  dependencies.push({ path, captured, bytes: bytes.length, sha256: hash(bytes) });
}
await writeFile(join(directory, label + '-' + stage + '-execution-sources.json'), JSON.stringify({ stage, copied, dependencies,
  actualDesktopExecuted: false, realModelRequests: 0,
  note: 'Captured before launch. Separate public RPC, Renderer and owned-stop receipts establish actual execution results.' }, null, 2), { flag: 'wx' });
console.log(JSON.stringify({ executionSourcesCaptured: true, stage, copied: copied.length, dependencies: dependencies.length, realModelRequests: 0 }));
